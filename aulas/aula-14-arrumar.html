import 'package:flutter/foundation.dart';

import 'forecast_repository.dart';
import 'island.dart';

/// The state of the islands screen. Knows the repository, not the widgets.
class ForecastNotifier extends ChangeNotifier {
  ForecastNotifier(this._repository); // injected, not created here

  final ForecastRepository _repository;

  bool isLoading = false;
  String? error;
  List<Island> islands = const [];

  Future<void> load() async {
    isLoading = true;
    error = null;
    notifyListeners(); // the screen shows the spinner
    try {
      islands = await _repository.fetchIslands();
    } catch (e) {
      error = '$e';
    } finally {
      isLoading = false;
      notifyListeners(); // the screen shows the islands or the error
    }
  }
}

  int get count => _ids.length;
  bool contains(String islandId) => _ids.contains(islandId);

    if (!_ids.remove(islandId)) _ids.add(islandId);
    notifyListeners();
  }
}

/// The favourite islands. Reads two notifiers.
class FavoritesScreen extends StatelessWidget {
  const FavoritesScreen({super.key});

  @override
  Widget build(BuildContext context) {
    final favorites = context.watch<FavoritesNotifier>();
    final islands = context
        .watch<ForecastNotifier>()
        .islands
        .where((island) => favorites.contains(island.id))
        .toList();
    return Scaffold(
      appBar: AppBar(title: const Text('Favoritas')),
      body: islands.isEmpty
          ? const Center(child: Text('Ainda não há ilhas favoritas.'))
          : ListView(
              children: [
                for (final island in islands)
                  ListTile(
                    title: Text(island.name),
                    trailing: FavoriteButton(islandId: island.id),
                  ),
              ],
            ),
    );
  }
}

import 'forecast.dart';
import 'island.dart';
import 'api_client.dart';
import 'forecast_repository.dart';

/// How: HTTP to the course API, through the ApiClient.
class HttpForecastRepository implements ForecastRepository {
  HttpForecastRepository(this._api); // injected, not created here

  final ApiClient _api;

  @override
  Future<List<Island>> fetchIslands() async {
    final list = await _api.get('/v1/islands') as List<dynamic>;
    return list
        .map((item) => Island.fromJson(item as Map<String, dynamic>))
        .toList();
  }

  @override
  Future<Forecast> fetchForecast(String islandId) async {
    final json = await _api.get('/v1/islands/$islandId/forecast');
    return Forecast.fromJson(json as Map<String, dynamic>);
  }
}

É o branch passo-2. Pergunta final: se amanhã a API mudar o JSON, em que pastas mexes? (models e data). Se mudar a cor da estrela? (widgets). Se a lista passar a ser uma grelha? (screens).

Onde mexes?

  • A API mudou o nome de um campo do JSON.
  • A estrela passa a ser um coração.
  • As favoritas passam a ficar guardadas no telemóvel.
  • Queremos testar a app sem rede.
Respostas. Campo do JSON: models/ (o fromJson). Coração: widgets/favorite_button.dart. Guardar as favoritas: state/favorites_notifier.dart (ou um repository novo em data/). Sem rede: data/fake_forecast_repository.dart e o main.dart com FAKE=true.
